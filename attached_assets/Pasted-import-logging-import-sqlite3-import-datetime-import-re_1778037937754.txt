# ====================== SEU TELEGRAM ID ======================
ADMIN_ID = 0   # ← TROQUE ESTE NÚMERO pelo seu ID (use /start para descobrir)

# ====================== BANCO DE DADOS ======================
def init_db():
    conn = sqlite3.connect('controle.db')
    c = conn.cursor()
    
    c.execute('''CREATE TABLE IF NOT EXISTS produtos (
                 codigo TEXT PRIMARY KEY, nome TEXT, estoque REAL, preco_venda REAL)''')
    
    c.execute('''CREATE TABLE IF NOT EXISTS pedidos (
                 id INTEGER PRIMARY KEY, numero TEXT, cliente TEXT, total REAL,
                 taxa REAL DEFAULT 0, pagamento TEXT, responsavel TEXT, data TEXT)''')
    
    c.execute('''CREATE TABLE IF NOT EXISTS itens_pedido (
                 pedido_id INTEGER, produto TEXT, quantidade REAL)''')
    
    c.execute('''CREATE TABLE IF NOT EXISTS retiradas (
                 id INTEGER PRIMARY KEY, socio TEXT, valor REAL, data TEXT)''')
    
    c.execute('''CREATE TABLE IF NOT EXISTS config (
                 chave TEXT PRIMARY KEY, valor REAL)''')
    
    # Produtos
    produtos = [
        ("ICE", "Ice o Lator", 375.0, 140.0),
        ("PAK", "Pak", 170.0, 60.0),
        ("CRUMBLE", "Crumble", 83.0, 180.0),
        ("POD", "Pod THC", 8.0, 450.0)
    ]
    c.executemany("INSERT OR IGNORE INTO produtos VALUES (?,?,?,?)", produtos)
    
    # Configurações iniciais
    c.execute("INSERT OR IGNORE INTO config VALUES ('saldo_banco', 0)")
    c.execute("INSERT OR IGNORE INTO config VALUES ('divida_fornecedor', 74890)")
    
    conn.commit()
    conn.close()

init_db()

def get_db():
    return sqlite3.connect('controle.db')

# ====================== PARSER DE PEDIDOS ======================
async def processar_pedido(update: Update, context: ContextTypes.DEFAULT_TYPE):
    texto = update.message.text.strip()
    if not re.search(r'pedido|total', texto, re.IGNORECASE):
        return

    try:
        lines = [line.strip() for line in texto.split('\n') if line.strip()]
        
        numero = re.search(r'pedido\s*(\d+)', lines[0], re.IGNORECASE)
        numero = numero.group(1) if numero else "MANUAL"
        
        cliente = lines[1] if len(lines) > 1 else "Desconhecido"
        
        itens = {}
        total = 0.0
        taxa = 0.0
        pagamento = "PIX"
        responsavel = "Não informado"
        
        for line in lines:
            line_u = line.upper()
            for prod in ["ICE", "PAK", "CRUMBLE", "POD"]:
                if prod in line_u:
                    q = re.search(r'(\d+)', line)
                    if q:
                        itens[prod] = itens.get(prod, 0) + float(q.group(1))
            
            if any(x in line_u for x in ["TOTAL", "R$"]):
                v = re.search(r'[\d\.,]+', line.replace(',', '.'))
                if v: total = float(v.group())
            
            if "TAXA" in line_u:
                v = re.search(r'[\d\.,]+', line.replace(',', '.'))
                if v: taxa = float(v.group())
            
            if "DINHEIRO" in line_u: pagamento = "DINHEIRO"
            if any(x in line_u for x in ["RD", "BART"]):
                responsavel = line.split(':')[-1].strip() if ':' in line else line.strip()
        
        if total <= 0:
            await update.message.reply_text("❌ Informe o Total do pedido.")
            return
        
        conn = get_db()
        c = conn.cursor()
        data = datetime.datetime.now().strftime("%Y-%m-%d %H:%M")
        
        c.execute("""INSERT INTO pedidos 
                     (numero, cliente, total, taxa, pagamento, responsavel, data) 
                     VALUES (?,?,?,?,?,?,?)""",
                  (numero, cliente, total, taxa, pagamento, responsavel, data))
        pedido_id = c.lastrowid
        
        for prod, qtd in itens.items():
            c.execute("INSERT INTO itens_pedido VALUES (?,?,?)", (pedido_id, prod, qtd))
            c.execute("UPDATE produtos SET estoque = estoque - ? WHERE codigo = ?", (qtd, prod))
        
        conn.commit()
        conn.close()
        
        await update.message.reply_text(f"✅ Pedido #{numero} registrado!\n👤 {cliente}\n💰 R$ {total:.2f} ({pagamento})")
        
    except Exception as e:
        await update.message.reply_text(f"❌ Erro ao registrar pedido.\nVerifique o formato.")

# ====================== RELATÓRIO DO DIA ======================
async def relatorio_dia(update: Update, context: ContextTypes.DEFAULT_TYPE):
    if not is_admin(update.effective_user.id):
        await update.message.reply_text("❌ Apenas o administrador pode ver o relatório.")
        return
    
    data_hoje = datetime.datetime.now().strftime("%Y-%m-%d")
    conn = get_db()
    c = conn.cursor()
    
    c.execute("SELECT SUM(total), COUNT(*) FROM pedidos WHERE data LIKE ?", (data_hoje+'%',))
    total_vendido, qtd_pedidos = c.fetchone() or (0, 0)
    
    c.execute("SELECT pagamento, SUM(total) FROM pedidos WHERE data LIKE ? GROUP BY pagamento", (data_hoje+'%',))
    pagamentos = dict(c.fetchall())
    
    c.execute("""SELECT i.produto, SUM(i.quantidade) 
                 FROM itens_pedido i JOIN pedidos p ON i.pedido_id = p.id 
                 WHERE p.data LIKE ? GROUP BY i.produto""", (data_hoje+'%',))
    saidas = dict(c.fetchall())
    
    c.execute("SELECT codigo, nome, estoque FROM produtos")
    estoque = c.fetchall()
    
    c.execute("SELECT valor FROM config WHERE chave='saldo_banco'")
    saldo_banco = c.fetchone()[0]
    
    c.execute("SELECT valor FROM config WHERE chave='divida_fornecedor'")
    divida = c.fetchone()[0]
    
    conn.close()
    
    rel = f"📦 **ENTREGAS DIA {datetime.datetime.now().strftime('%d/%m/%Y')}**\n"
    rel += "━━━━━━━━━━━━━━━━━━\n\n"
    rel += f"Pedidos: {qtd_pedidos}\n"
    rel += f"💰 Total Vendido: R$ {total_vendido:.2f}\n"
    rel += f"📲 PIX: R$ {pagamentos.get('PIX', 0):.2f}\n"
    rel += f"💵 Dinheiro: R$ {pagamentos.get('DINHEIRO', 0):.2f}\n\n"
    
    rel += "📦 **SAÍDAS DO DIA**\n"
    for p in ["ICE", "PAK", "CRUMBLE", "POD"]:
        rel += f"{p}: {saidas.get(p, 0):.1f}g\n"
    
    rel += "\n📦 **ESTOQUE RESTANTE**\n"
    for cod, nome, qtd in estoque:
        rel += f"{cod} ({nome}): {qtd:.1f}\n"
    
    rel += f"\n🏦 Saldo Banco: R$ {saldo_banco:.2f}\n"
    rel += f"📉 Dívida Fornecedor: R$ {divida:.2f}\n"
    rel += "━━━━━━━━━━━━━━━━━━"
    
    await update.message.reply_text(rel, parse_mode='Markdown')

async def start(update: Update, context: ContextTypes.DEFAULT_TYPE):

    await update.message.reply_text(
        f"🔑 Seu ID: <code>{uid}</code>\n\n"
        "Coloque esse número na linha `ADMIN_ID` do código.\n\n"
        "Use /relatorio no final do dia.", 
        parse_mode='HTML'
    )

async def retirada(update: Update, context: ContextTypes.DEFAULT_TYPE):
    if not is_admin(update.effective_user.id):
        await update.message.reply_text("❌ Apenas o dono pode registrar retiradas.")
        return
    try:
        socio = "RD" if update.message.text.lower().startswith('/rd') else "Bart"
        valor = float(context.args[0].replace(',', '.'))
        conn = get_db()
        c = conn.cursor()
        data = datetime.datetime.now().strftime("%Y-%m-%d %H:%M")
        c.execute("INSERT INTO retiradas (socio, valor, data) VALUES (?,?,?)", (socio, valor, data))
        conn.commit()
        conn.close()
        await update.message.reply_text(f"✅ {socio} retirou R$ {valor:.2f}")
    except:
        await update.message.reply_text("Uso: /rd 500 ou /bart 300")

async def banco(update: Update, context: ContextTypes.DEFAULT_TYPE):
