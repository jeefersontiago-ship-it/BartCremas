import logging
import sqlite3
import datetime
import re
from telegram import Update
from telegram.ext import ApplicationBuilder, CommandHandler, MessageHandler, filters, ContextTypes

logging.basicConfig(level=logging.INFO)

TOKEN = "SEU_TOKEN_AQUI"

# ====================== BANCO ======================
def init_db():
    conn = sqlite3.connect('controle.db')
    c = conn.cursor()
    
    c.execute('''CREATE TABLE IF NOT EXISTS produtos (
                 codigo TEXT PRIMARY KEY, nome TEXT, estoque REAL, preco_venda REAL)''')
    
    c.execute('''CREATE TABLE IF NOT EXISTS pedidos (
                 id INTEGER PRIMARY KEY,
                 numero TEXT,
                 cliente TEXT,
                 total REAL,
                 taxa REAL DEFAULT 0,
                 pagamento TEXT,        -- PIX ou DINHEIRO
                 responsavel TEXT,
                 data TEXT)''')
    
    c.execute('''CREATE TABLE IF NOT EXISTS itens_pedido (
                 pedido_id INTEGER,
                 produto TEXT,
                 quantidade REAL)''')
    
    # Produtos iniciais
    produtos = [
        ("ICE", "Ice o Lator", 375, 140),
        ("PAK", "Pak", 170, 60),
        ("CRUMBLE", "Crumble", 83, 180),
        ("POD", "Pod THC", 8, 450)
    ]
    c.executemany("INSERT OR IGNORE INTO produtos VALUES (?,?,?,?)", produtos)
    conn.commit()
    conn.close()

init_db()

def get_db():
    return sqlite3.connect('controle.db')

# ====================== PARSER MELHORADO ======================
async def processar_pedido(update: Update, context: ContextTypes.DEFAULT_TYPE):
    texto = update.message.text.strip()
    try:
        lines = [line.strip() for line in texto.split('\n') if line.strip()]
        
        numero = re.search(r'pedido\s*(\d+)', lines[0], re.IGNORECASE)
        numero = numero.group(1) if numero else datetime.datetime.now().strftime("%d%H%M")
        
        cliente = lines[1] if len(lines) > 1 else "Desconhecido"
        
        itens = {}
        total = 0.0
        taxa = 0.0
        pagamento = "PIX"          # padrão
        responsavel = "Não informado"
        
        for line in lines:
            line_u = line.upper()
            
            # Produtos
            for prod in ["ICE", "PAK", "CRUMBLE", "POD"]:
                if prod in line_u:
                    q = re.search(r'(\d+)', line)
                    if q:
                        itens[prod] = itens.get(prod, 0) + float(q.group(1))
            
            # Total
            if any(x in line_u for x in ["TOTAL", "R$"]):
                v = re.search(r'[\d\.,]+', line.replace(',', '.'))
                if v:
                    total = float(v.group())
            
            # Taxa
            if "TAXA" in line_u:
                v = re.search(r'[\d\.,]+', line.replace(',', '.'))
                if v: taxa = float(v.group())
            
            # Pagamento
            if "DINHEIRO" in line_u or "GRANA" in line_u or "REAL" in line_u:
                pagamento = "DINHEIRO"
            if "PIX" in line_u:
                pagamento = "PIX"
            
            # Responsável
            if any(x in line_u for x in ["RD", "BART", "RESPONSAVEL", "RESPONSÁVEL"]):
                responsavel = line.split(':')[-1].strip() if ':' in line else line.split()[-1]
        
        if total <= 0:
            await update.message.reply_text("❌ Informe o Total (ex: Total: R$ 440)")
            return
        
        # Salvar
        conn = get_db()
        c = conn.cursor()
        data = datetime.datetime.now().strftime("%Y-%m-%d %H:%M")
        
        c.execute("""INSERT INTO pedidos 
                     (numero, cliente, total, taxa, pagamento, responsavel, data) 
                     VALUES (?,?,?,?,?,?,?)""",
                  (numero, cliente, total, taxa, pagamento, responsavel, data))
        pedido_id = c.lastrowid
        
        for prod, qtd in itens.items():
            c.execute("INSERT INTO itens_pedido VALUES (?,?,?)", (pedido_id, prod, qtd))
            c.execute("UPDATE produtos SET estoque = estoque - ? WHERE codigo = ?", (qtd, prod))
        
        conn.commit()
        conn.close()
        
        await update.message.reply_text(f"✅ Pedido #{numero} registrado!\n👤 {cliente}\n💰 R$ {total:.2f} ({pagamento})")
        
    except Exception as e:
        await update.message.reply_text(f"❌ Erro: {str(e)}\n\nTente novamente.")

# ====================== RELATÓRIO FINAL DO DIA ======================
async def relatorio_dia(update: Update, context: ContextTypes.DEFAULT_TYPE):
    data_hoje = datetime.datetime.now().strftime("%Y-%m-%d")
    
    conn = get_db()
    c = conn.cursor()
    
    # Totais
    c.execute("SELECT SUM(total), COUNT(*) FROM pedidos WHERE data LIKE ?", (data_hoje + '%',))
    total_dia, qtd_pedidos = c.fetchone()
    
    c.execute("SELECT pagamento, SUM(total) FROM pedidos WHERE data LIKE ? GROUP BY pagamento", (data_hoje + '%',))
    pagamentos = dict(c.fetchall())
    
    # Saídas por produto
    c.execute("""SELECT i.produto, SUM(i.quantidade) 
                 FROM itens_pedido i 
                 JOIN pedidos p ON i.pedido_id = p.id 
                 WHERE p.data LIKE ? 
                 GROUP BY i.produto""", (data_hoje + '%',))
    saidas = dict(c.fetchall())
    
    # Estoque atual
    c.execute("SELECT codigo, nome, estoque FROM produtos")
    estoque_atual = c.fetchall()
    
    conn.close()
    
    # Montagem do relatório bonito
    rel = f"📦 **ENTREGAS DIA {datetime.datetime.now().strftime('%d/%m/%Y')}**\n"
    rel += "━━━━━━━━━━━━━━━━━━\n\n"
    
    rel += f"Total de pedidos: {qtd_pedidos}\n"
    rel += f"💰 Total Entrada: R$ {total_dia or 0:.2f}\n"
    rel += f"📲 PIX: R$ {pagamentos.get('PIX', 0):.2f}\n"
    rel += f"💵 Dinheiro: R$ {pagamentos.get('DINHEIRO', 0):.2f}\n\n"
    
    rel += "📦 **SAÍDAS DO DIA**\n"
    for cod in ["ICE", "PAK", "CRUMBLE", "POD"]:
        saida = saidas.get(cod, 0)
        rel += f"{cod}: {saida:.1f}g\n"
    
    rel += "\n📦 **ESTOQUE RESTANTE**\n"
    for cod, nome, qtd in estoque_atual:
        rel += f"{cod} ({nome}): {qtd:.1f}\n"
    
    rel += "\n━━━━━━━━━━━━━━━━━━"
    await update.message.reply_text(rel, parse_mode='Markdown')

# ====================== COMANDOS ======================
async def start(update: Update, context: ContextTypes.DEFAULT_TYPE):
    await update.message.reply_text(
        "🍪 <b>Sistema Ativo</b>\n\n"
        "Cole o pedido assim:\n\n"
        "<code>pedido 01\n"
        "Poliana\n"
        "5G - PAK\n"
        "1G - ICE\n"
        "Total: R$ 440\n"
        "Dinheiro\n"
        "Responsavel: RD</code>\n\n"
        "Use /relatorio no final do dia",
        parse_mode='HTML'
    )

async def estoque(update: Update, context: ContextTypes.DEFAULT_TYPE):
    conn = get_db()
    c = conn.cursor()
    c.execute("SELECT codigo, nome, estoque FROM produtos")
    rows = c.fetchall()
    conn.close()
    
    msg = "📦 <b>ESTOQUE ATUAL</b>\n\n"
    for cod, nome, qtd in rows:
        msg += f"<b>{cod}</b> - {nome}: {qtd:.1f}\n"
    await update.message.reply_text(msg, parse_mode='HTML')

# ====================== INICIAR ======================
app = ApplicationBuilder().token(TOKEN).build()

app.add_handler(CommandHandler("start", start))
app.add_handler(CommandHandler("estoque", estoque))
app.add_handler(CommandHandler("relatorio", relatorio_dia))
app.add_handler(CommandHandler("fechamento", relatorio_dia))
app.add_handler(MessageHandler(filters.TEXT & ~filters.COMMAND, processar_pedido))

print("🚀 Bot Profissional Rodando!")
app.run_polling()
