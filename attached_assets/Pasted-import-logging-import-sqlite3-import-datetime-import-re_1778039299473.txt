# ====================== BANCO ======================
def init_db():
    conn = sqlite3.connect('controle.db')
    c = conn.cursor()
    
    c.execute('''CREATE TABLE IF NOT EXISTS produtos (
                 codigo TEXT PRIMARY KEY, nome TEXT, estoque REAL, preco REAL)''')
    
    c.execute('''CREATE TABLE IF NOT EXISTS pedidos (
                 id INTEGER PRIMARY KEY, numero TEXT, cliente TEXT, total REAL,
                 taxa REAL, pagamento TEXT DEFAULT 'PIX', responsavel TEXT, data TEXT)''')
    
    c.execute('''CREATE TABLE IF NOT EXISTS itens_pedido (
                 pedido_id INTEGER, produto TEXT, quantidade REAL)''')
    
    produtos = [
        ("ICE", "🍦 Ice o Lator", 375.0, 140.0),
        ("PAK", "🥐 Pak", 170.0, 60.0),
        ("CRUMBLE", "🍪 Crumble", 83.0, 180.0),
        ("POD", "🪦 Pod THC", 8.0, 450.0)
    ]
    c.executemany("INSERT OR IGNORE INTO produtos VALUES (?,?,?,?)", produtos)
    conn.commit()
    conn.close()

init_db()
def get_db(): return sqlite3.connect('controle.db')

# ====================== MENU ======================
async def start(update: Update, context: ContextTypes.DEFAULT_TYPE):
    keyboard = [
        [InlineKeyboardButton("📋 Novo Pedido", callback_data="novo_pedido")],
        [InlineKeyboardButton("📦 Ver Estoque", callback_data="estoque")],
        [InlineKeyboardButton("📊 Relatório do Dia", callback_data="relatorio")],
    ]
    reply_markup = InlineKeyboardMarkup(keyboard)
    
    await update.message.reply_text(
        "🍪 <b>Cookie Control Pro</b>\n"
        "━━━━━━━━━━━━━━━━━━\n"
        "Sistema com cálculo automático + mensagem pro cliente",
        parse_mode='HTML', reply_markup=reply_markup
    )

# ====================== FLUXO DO PEDIDO ======================
async def button_callback(update: Update, context: ContextTypes.DEFAULT_TYPE):
    query = update.callback_query
    await query.answer()

    if query.data == "novo_pedido":
        await query.edit_message_text("👤 Digite o nome do cliente:")
        context.user_data['estado'] = 'esperando_cliente'

async def handle_message(update: Update, context: ContextTypes.DEFAULT_TYPE):
    estado = context.user_data.get('estado')
    if not estado: return

    if estado == 'esperando_cliente':
        context.user_data['cliente'] = update.message.text
        context.user_data['estado'] = 'esperando_itens'
        await update.message.reply_text("🛒 Envie os itens (um por linha):\nEx: 5 PAK\n2 ICE")

    elif estado == 'esperando_itens':
        await calcular_pedido(update, context, update.message.text)

    elif estado == 'confirmar':
        if update.message.text.lower() in ['sim', 'ok', 'confirmar']:
            await salvar_pedido(update, context)
        else:
            await update.message.reply_text("Pedido cancelado.")
            context.user_data.clear()

# ====================== CÁLCULO + TAXA ======================
async def calcular_pedido(update: Update, context: ContextTypes.DEFAULT_TYPE, texto):
    try:
        conn = get_db()
        c = conn.cursor()
        itens = {}
        subtotal = 0.0

        for line in texto.upper().split('\n'):
            match = re.search(r'(\d+[\.,]?\d*)\s*(ICE|PAK|CRUMBLE|POD)', line)
            if match:
                qtd = float(match.group(1).replace(',', '.'))
                prod = match.group(2)
                itens[prod] = qtd
                c.execute("SELECT preco FROM produtos WHERE codigo=?", (prod,))
                preco = c.fetchone()[0]
                subtotal += qtd * preco

        taxa = 10.0 if subtotal < 500 else 0.0
        total_final = subtotal + taxa

        conn.close()

        context.user_data['itens'] = itens
        context.user_data['subtotal'] = subtotal
        context.user_data['taxa'] = taxa
        context.user_data['total'] = total_final
        context.user_data['estado'] = 'confirmar'

        msg = "📋 <b>RESUMO DO PEDIDO</b>\n\n"
        msg += f"👤 Cliente: {context.user_data['cliente']}\n\n"
        for p, q in itens.items():
            msg += f"• {q:.1f} × {p}\n"
        msg += f"\n💰 Subtotal: R$ {subtotal:.2f}"
        if taxa > 0:
            msg += f"\n📌 Taxa entrega: R$ {taxa:.2f}"
        msg += f"\n\n💎 <b>Total: R$ {total_final:.2f}</b>\n\n"
        msg += "Responda **Sim** para confirmar e gerar mensagem pro cliente."

        await update.message.reply_text(msg, parse_mode='HTML')

    except Exception as e:
        await update.message.reply_text("Erro ao calcular.")

# ====================== SALVAR + MENSAGEM PRO CLIENTE ======================
async def salvar_pedido(update: Update, context: ContextTypes.DEFAULT_TYPE):
    try:
        conn = get_db()
        c = conn.cursor()
        data = datetime.datetime.now().strftime("%Y-%m-%d %H:%M")
        numero = datetime.datetime.now().strftime("%d%H%M")

        c.execute("""INSERT INTO pedidos 
                     (numero, cliente, total, taxa, pagamento, responsavel, data)
                     VALUES (?,?,?,?,?,?,?)""",
                  (numero, context.user_data['cliente'], context.user_data['total'],
                   context.user_data['taxa'], 'PIX', 'Você', data))
        pedido_id = c.lastrowid

        for prod, qtd in context.user_data['itens'].items():
            c.execute("INSERT INTO itens_pedido VALUES (?,?,?)", (pedido_id, prod, qtd))
            c.execute("UPDATE produtos SET estoque = estoque - ? WHERE codigo = ?", (qtd, prod))

        conn.commit()
        conn.close()

        # ================= MENSAGEM PARA O CLIENTE =================
        cliente_msg = f"✅ *Pedido Confirmado!*\n\n"
        cliente_msg += f"Olá *{context.user_data['cliente']}*,\n\n"
        cliente_msg += "Seu pedido foi registrado:\n\n"
        
        for p, q in context.user_data['itens'].items():
            cliente_msg += f"• {q:.1f} × {p}\n"
        
        cliente_msg += f"\n💰 *Total a pagar: R$ {context.user_data['total']:.2f}*\n\n"
        if context.user_data['taxa'] > 0:
            cliente_msg += "📌 Taxa de entrega inclusa\n\n"
        
        cliente_msg += f"🔑 *Chave PIX:* `{CHAVE_PIX}`\n"
        cliente_msg += f"💵 Valor exato: R$ {context.user_data['total']:.2f}\n\n"
        cliente_msg += "Por favor, envie o comprovante após o pagamento."

        await update.message.reply_text(
            "✅ Pedido salvo com sucesso!\n\n"
            "📄 **Mensagem pronta para o cliente:**\n\n"
            + cliente_msg,
            parse_mode='Markdown'
        )

        context.user_data.clear()

    except Exception as e:
        await update.message.reply_text(f"Erro ao salvar: {e}")
