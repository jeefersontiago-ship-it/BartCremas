# ====================== RESET COMANDOS ======================
async def reset_dia(update: Update, context: ContextTypes.DEFAULT_TYPE):
    if not is_admin(update.effective_user.id):
        await update.message.reply_text("❌ Apenas o administrador pode resetar.")
        return

    data_hoje = datetime.datetime.now().strftime("%Y-%m-%d")
    conn = get_db()
    c = conn.cursor()
    
    # Deletar pedidos e itens do dia
    c.execute("DELETE FROM pedidos WHERE data LIKE ?", (data_hoje + '%',))
    c.execute("DELETE FROM itens_pedido WHERE pedido_id NOT IN (SELECT id FROM pedidos)")
    
    conn.commit()
    conn.close()
    
    await update.message.reply_text(
        "🗑️ **Reset do dia realizado com sucesso!**\n\n"
        "Todos os pedidos, totais e saídas de hoje foram apagados.\n"
        "O estoque **não** foi alterado."
    )

async def reset_completo(update: Update, context: ContextTypes.DEFAULT_TYPE):
    if not is_admin(update.effective_user.id):
        await update.message.reply_text("❌ Acesso negado.")
        return
    
    if not context.args or context.args[0].lower() != "confirmar":
        await update.message.reply_text(
            "⚠️ **CUIDADO!** Isso vai limpar TODOS os pedidos.\n\n"
            "Digite `/resetcompleto confirmar` para confirmar."
        )
        return

    conn = get_db()
    c = conn.cursor()
    c.execute("DELETE FROM pedidos")
    c.execute("DELETE FROM itens_pedido")
    conn.commit()
    conn.close()
    
    await update.message.reply_text("🗑️ **Reset completo realizado!** Todos os históricos de pedidos foram apagados.")
